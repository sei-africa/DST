The coordinate in the GeoJSON must be EPSG:4326 (WGS84). The geometry types should be Point, Polygon or MultiPolygon. You can have one or multiple geometry types in one file.

The data structures are as follow 

{
    "type": "FeatureCollection",
    "features": [
    {
        "type": "Feature",
        "geometry":
        {
            "type": "Point",
            "coordinates": [35.3223, -15.38361]
        },
        "properties":
        {
            "type": "Point",
            "name": "Ahi Private Clinic",
            .......
        }
    },
    {
        "type": "Feature",
        "geometry":
        {
            "type": "Polygon",
            "coordinates": [
                [
                    [34.8341, -15.2765],
                    [34.835, -15.2762],

                    .......

                    [34.8327, -15.2769],
                    [34.8341, -15.2765]
                ]
            ]
        },
        "properties":
        {
            "type": "Polygon",
            "name": "Balaka-DHO",
            .......
        }
    },
    { 
        .......... 
    },

    ..........
    ]
}
